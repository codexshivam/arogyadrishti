# --- HELPER: Fetch District Representative & Village Staff from API or Google Sheets ---
def get_authorized_contacts(ward_village: str, district_lgd: str):
    sheet = gc.open_by_key(SPREADSHEET_ID).worksheet("3_Staff_Directory")
    records = sheet.get_all_records()

    # 1. Match Village Officer & District Rep in Google Sheets first (allows instant admin override)
    for row in records:
        if str(row["Ward_or_Village"]).strip().lower() == ward_village.strip().lower():
            return {
                "village_officer_name": row["Village_Officer_Name"],
                "village_officer_email": row["Village_Officer_Email"],
                "village_officer_phone": row["Village_Officer_Phone"],
                "district_rep_name": row.get("District_Rep_Name"),
                "district_rep_phone": row.get("District_Rep_Phone"),
                "district_rep_email": row.get("District_Rep_Email"),
            }

    # 2. Fallback to data.gov.in OGD Public API for District Representative if not in Sheet
    ogd_url = f"https://api.data.gov.in/resource/district-health-directory?api-key={OGD_API_KEY}&format=json&filters[lgd_code]={district_lgd}"
    resp = requests.get(ogd_url, timeout=5)
    if resp.status_code == 200 and resp.json().get("records"):
        rec = resp.json()["records"][0]
        return {
            "village_officer_name": "Block Medical Officer",
            "village_officer_email": rec.get("bmo_email", "nodal.nvbdcp@gov.in"),
            "village_officer_phone": rec.get("bmo_phone", ""),
            "district_rep_name": rec.get("cmo_name", "District Malaria Officer"),
            "district_rep_phone": rec.get("cmo_phone", "104"),
            "district_rep_email": rec.get("cmo_email", "dmo@gov.in")
        }
    return None

# --- HELPER: Google Maps Routes API for Nearest Van & Traffic ETA ---
def assign_nearest_van_with_eta(target_lat: float, target_lng: float):
    vans_sheet = gc.open_by_key(SPREADSHEET_ID).worksheet("1_Vans_Fleet")
    vans = vans_sheet.get_all_records()

    best_van = None
    min_duration_sec = float("inf")

    for idx, van in enumerate(vans, start=2): # Row 1 is header
        if van["Status"] != "Available":
            continue

        # Call Google Maps Routes API (computeRoutes)
        routes_url = "https://routes.googleapis.com/directions/v2:computeRoutes"
        headers = {
            "Content-Type": "application/json",
            "X-Goog-Api-Key": MAPS_API_KEY,
            "X-Goog-FieldMask": "routes.duration,routes.distanceMeters"
        }
        body = {
            "origin": {"location": {"latLng": {"latitude": float(van["Base_Lat"]), "longitude": float(van["Base_Lng"])}}},
            "destination": {"location": {"latLng": {"latitude": target_lat, "longitude": target_lng}}},
            "travelMode": "DRIVE",
            "routingPreference": "TRAFFIC_AWARE"
        }
        r = requests.post(routes_url, json=body, headers=headers).json()
        if "routes" in r and r["routes"]:
            duration_sec = int(r["routes"][0]["duration"].replace("s", ""))
            distance_km = round(r["routes"][0]["distanceMeters"] / 1000, 2)
            if duration_sec < min_duration_sec:
                min_duration_sec = duration_sec
                best_van = {
                    "row_idx": idx,
                    "van_id": van["Van_ID"],
                    "vehicle_no": van["Vehicle_No"],
                    "driver_name": van["Driver_Name"],
                    "driver_phone": van["Driver_Phone"],
                    "eta_mins": round(duration_sec / 60),
                    "distance_km": distance_km
                }

    if best_van:
        # Mark Van as Dispatched in Google Sheet
        vans_sheet.update_cell(best_van["row_idx"], 7, "Dispatched")
    return best_van

@app.post("/api/v1/dispatch-action")
def handle_modal_action(payload: ActionPayload):
    contacts = get_authorized_contacts(payload.ward_village, payload.district_lgd)

    # 1. Log audit trail in Cloud Firestore
    doc_ref = db.collection("dispatches").document()
    firestore_record = {
        "officer_uid": payload.officer_uid,
        "action_type": payload.action_type,
        "ward_village": payload.ward_village,
        "location": firestore.GeoPoint(payload.latitude, payload.longitude),
        "notes": payload.notes,
        "timestamp": firestore.SERVER_TIMESTAMP,
        "contacts_notified": contacts
    }

    if payload.action_type == "REQUEST_VAN":
        assigned_van = assign_nearest_van_with_eta(payload.latitude, payload.longitude)
        if not assigned_van:
            raise HTTPException(status_code=404, detail="No available fogging vans found.")

        # Append to '4_Van_Requests' Google Sheet
        req_sheet = gc.open_by_key(SPREADSHEET_ID).worksheet("4_Van_Requests")
        req_sheet.append_row([
            doc_ref.id,
            str(firestore.SERVER_TIMESTAMP),
            payload.officer_uid,
            payload.ward_village,
            payload.latitude,
            payload.longitude,
            assigned_van["vehicle_no"],
            assigned_van["driver_phone"],
            f"{assigned_van['eta_mins']} mins",
            f"{assigned_van['distance_km']} km",
            "Dispatched"
        ])

        firestore_record["assigned_van"] = assigned_van
        doc_ref.set(firestore_record)
        return {"status": "VAN_ASSIGNED", "van_details": assigned_van}

    elif payload.action_type == "SEND_ALERT":
        # Append to '5_Alert_Logs' Google Sheet
        alert_sheet = gc.open_by_key(SPREADSHEET_ID).worksheet("5_Alert_Logs")
        alert_sheet.append_row([
            doc_ref.id,
            str(firestore.SERVER_TIMESTAMP),
            payload.ward_village,
            "CRITICAL",
            payload.notes,
            contacts["village_officer_email"] if contacts else "N/A",
            contacts["district_rep_phone"] if contacts else "N/A"
        ])
        doc_ref.set(firestore_record)

        # Trigger Automatic Email to Village Staff & District Representative
        if contacts and contacts["village_officer_email"]:
            send_emergency_email(
                to_email=contacts["village_officer_email"],
                cc_email=contacts.get("district_rep_email"),
                ward=payload.ward_village,
                lat=payload.latitude,
                lng=payload.longitude,
                notes=payload.notes,
                district_rep_phone=contacts.get("district_rep_phone")
            )
        return {"status": "ALERT_SENT", "notified": contacts}
5. Public Prediction API (GET /api/v1/risk-index?lat=&lng=) with Google Earth EngineThis public DPI endpoint allows any citizen app, hospital, or state portal across India to pass lat and lng and get a scientifically backed vector risk forecast:Python@app.get("/api/v1/risk-index")
def get_public_vector_risk(lat: float, lng: float):
    point = ee.Geometry.Point([lng, lat])

    # 1. Query MODIS Land Surface Temperature (Last 7 days mean in Celsius)
    lst_img = (ee.ImageCollection("MODIS/061/MOD11A1")
               .filterDate("2026-10-15", "2026-10-22")
               .select("LST_Day_1km")
               .mean()
               .multiply(0.02)
               .subtract(273.15))

    # 2. Query Sentinel-2 NDWI (Surface Stagnant Water Index)
    s2_img = (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
              .filterBounds(point)
              .filterDate("2026-09-22", "2026-10-22")
              .sort("CLOUDY_PIXEL_PERCENTAGE")
              .first())
    ndwi = s2_img.normalizedDifference(["B3", "B8"]).rename("NDWI")

    # 3. Query CHIRPS 14-Day Cumulative Precipitation (mm)
    rain_img = (ee.ImageCollection("UCSB-CHG/CHIRPS/DAILY")
                .filterDate("2026-10-08", "2026-10-22")
                .sum())

    # Extract values at coordinate
    lst_val = lst_img.sample(point, 1000).first().get("LST_Day_1km").getInfo() or 28.5
    ndwi_val = ndwi.sample(point, 10).first().get("NDWI").getInfo() or 0.15
    rain_val = rain_img.sample(point, 5000).first().get("precipitation").getInfo() or 42.0

    # 4. Fetch Ground-Truth Breteau Index from Firestore for this Ward
    # (Combines satellite suitability with ASHA worker field logs)
    thermal_score = 1.0 if (26.0 <= lst_val <= 32.0) else 0.4
    water_score = min(max((ndwi_val + 0.2) / 0.6, 0.0), 1.0)
    rain_score = 1.0 if (50.0 <= rain_val <= 150.0) else 0.5

    composite_risk = round((0.4 * thermal_score) + (0.35 * water_score) + (0.25 * rain_score), 2)
    risk_tier = "CRITICAL" if composite_risk > 0.75 else ("HIGH" if composite_risk > 0.5 else "MODERATE")

    return {
        "coordinates": {"latitude": lat, "longitude": lng},
        "risk_tier": risk_tier,
        "composite_risk_score": composite_risk,
        "satellite_telemetry_gee": {
            "land_surface_temp_celsius": round(lst_val, 2),
            "stagnant_water_ndwi": round(ndwi_val, 3),
            "rainfall_14d_mm": round(rain_val, 1),
            "optimal_breeding_window": 26.0 <= lst_val <= 32.0
        },
        "recommended_intervention": "Immediate Temephos (1 ppm) larviciding & Pyrethrum space spray within 500m radius." if risk_tier == "CRITICAL" else "Routine weekly source reduction."
    }
